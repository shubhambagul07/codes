#include <xc.h>
#define _XTAL_FREQ 20000000

void uart_init() {
    TRISC6 = 0; TRISC7 = 1;
    SPBRG = 129;   // 9600 baud
    BRGH = 1;
    SYNC = 0; SPEN = 1;
    TXEN = 1; CREN = 1;
}

void uart_send(char data) {
    while(!TXIF);
    TXREG = data;
}

void main(void) {
    uart_init();
    uart_send('A');  // Send data to PC
    while(1);
}
