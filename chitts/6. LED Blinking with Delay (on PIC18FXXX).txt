#include <xc.h>
#define _XTAL_FREQ 20000000

void main(void) {
    TRISB = 0x00;     // Set PORTB as output
    LATB = 0x00;      // Clear PORTB

    while(1) {
        LATB = 0xFF;  // Turn all LEDs ON
        __delay_ms(500);
        LATB = 0x00;  // Turn all LEDs OFF
        __delay_ms(500);
    }
}
